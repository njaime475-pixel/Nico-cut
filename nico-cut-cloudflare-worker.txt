export default {
  async fetch(request, env) {
    const corsHeaders = {
      "Access-Control-Allow-Origin": "*",
      "Access-Control-Allow-Methods": "POST, OPTIONS",
      "Access-Control-Allow-Headers": "Content-Type",
      "Content-Type": "application/json",
    };

    if (request.method === "OPTIONS") {
      return new Response(null, { headers: corsHeaders });
    }

    if (request.method !== "POST") {
      return new Response(
        JSON.stringify({ ok: false, error: "Usá POST." }),
        { status: 405, headers: corsHeaders }
      );
    }

    try {
      const body = await request.json();
      const image = body.image;

      if (!image) {
        return new Response(
          JSON.stringify({ ok: false, error: "Falta la imagen." }),
          { status: 400, headers: corsHeaders }
        );
      }

      const prompt = `
Analizá esta foto de una comida como asistente nutricional.

Identificá los alimentos visibles y estimá sus cantidades en gramos.
Luego estimá calorías, proteínas, carbohidratos, grasas y fibra.

IMPORTANTE:
- Una foto no permite conocer pesos exactos. Hacé una estimación razonable.
- No inventes ingredientes que no sean visibles salvo que sean muy probables.
- Si hay incertidumbre, indicála.
- Respondé SOLO JSON válido, sin markdown ni texto adicional.

Formato exacto:
{
  "items": [
    {
      "name": "nombre del alimento",
      "grams": 150,
      "kcal": 250,
      "protein": 30,
      "carbs": 10,
      "fat": 8,
      "fiber": 2
    }
  ],
  "totals": {
    "kcal": 500,
    "protein": 45,
    "carbs": 50,
    "fat": 15,
    "fiber": 6
  },
  "confidence": "media",
  "note": "breve aclaración si hace falta"
}
`;

      const result = await env.AI.run(
        "@cf/meta/llama-3.2-11b-vision-instruct",
        {
          messages: [
            {
              role: "system",
              content:
                "Sos un asistente especializado en estimación nutricional visual.",
            },
            {
              role: "user",
              content: prompt,
            },
          ],
          image,
          max_tokens: 700,
        }
      );

      let text = result.response ?? result.result ?? result;

      if (typeof text !== "string") {
        text = JSON.stringify(text);
      }

      text = text
        .replace(/^```json\s*/i, "")
        .replace(/^```\s*/i, "")
        .replace(/\s*```$/i, "")
        .trim();

      let parsed;

      try {
        parsed = JSON.parse(text);
      } catch {
        parsed = {
          raw: text,
          warning: "La IA respondió, pero el JSON no pudo interpretarse.",
        };
      }

      return new Response(
        JSON.stringify({
          ok: true,
          analysis: parsed,
        }),
        { headers: corsHeaders }
      );
    } catch (err) {
      return new Response(
        JSON.stringify({
          ok: false,
          error: err?.message || "Error analizando la imagen.",
        }),
        { status: 500, headers: corsHeaders }
      );
    }
  },
};
